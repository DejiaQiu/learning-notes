1.安装

参考[https://github.com/doreamon-design/clash]
或者从包里下

2.配置
```bash
mkdir -p ~/.config/clash

wget -O ~/.config/clash/config.yaml "订阅链接"
```
3.运行
```bash
./clash -d ~/.config/clash
```
若没网，可将数据库复制到~/.config/clash

4.配置环境
```bash
echo "export http_proxy=http://127.0.0.1:7890" >> ~/.bashrc
echo "export https_proxy=http://127.0.0.1:7890" >> ~/.bashrc
source ~/.bashrc  # 使其生效
```

5.部署为系统服务
```bash
sudo nano /etc/systemd/system/clash.service
```

```bash
[Unit]
Description=Clash Service
After=network.target

[Service]
Type=simple
User=room
ExecStart=/home/room/software/clash/clash -d /home/room/.config/clash
Restart=on-failure
RestartSec=5s

[Install]
WantedBy=multi-user.target
```
设置服务并启动
```bash
sudo systemctl daemon-reload
sudo systemctl enable clash
sudo systemctl start clash
```
查看服务状态
```bash
sudo systemctl status clash
```
查看日志
```bash
 journalctl -u clash -f 
 ```
选择节点
```bash
curl -X PUT http://127.0.0.1:9090/proxies/🔰%20选择节点 \
     -H "Content-Type: application/json" \
     -d '{"name": "🇭🇰 香港Y02 | IEPL"}'
```


环境变量设置
```bash
unset http_proxy https_proxy ALL_PROXY

export http_proxy=http://127.0.0.1:7890
export https_proxy=http://127.0.0.1:7890
export ALL_PROXY=socks5://127.0.0.1:7891

```
